--1. Creare baza de date OLTP
create table LOCATIE
(
id_locatie int,
judet varchar(20),
localitate varchar(20),
strada varchar(20),
detalii varchar(50)
);

alter table LOCATIE
add constraint id_locatie_pk primary key(id_locatie);

create table UTILAJ
(
id_utilaj int,
nume varchar(50) not null, 
an_fabricatie date constraint an_fabricatie not null
);
alter table UTILAJ
add constraint id_utilaj_pk primary key(id_utilaj);

create table COLABORATORI
(
id_colaborator int,
nume varchar(100) not null, 
numar_telefon varchar(15) not null,
procent int default 10
);

alter table COLABORATORI
add constraint id_colaborator_pk primary key(id_colaborator);

create table CLIENT
(
id_client int,
nume varchar(100) not null, 
prenume varchar(100) not null, 
numar_telefon varchar(15) not null,
nr_lucrari_anterioare int
constraint nr_lucrari_anterioare_valid check(nr_lucrari_anterioare>=0)
);

alter table CLIENT
add constraint id_client_pk primary key(id_client);

create table OFERTA
(
id_oferta int,
pret int,
cod_colaborator int
);

alter table OFERTA
add constraint id_oferta_pk primary key(id_oferta);
alter table OFERTA
drop column cod_colaborator;

create table SEF
(
id_sef int, 
nume varchar(100) not null, 
prenume varchar(100) not null,
numar_telefon varchar(15) not null, 
varsta  int constraint varsta_valid check(varsta>=18),
experienta int constraint experienta_valid check(experienta>=0), 
salariu int constraint salariu_valid check(salariu>0)
);
alter table SEF
add constraint id_sef_pk primary key(id_sef);

create table LUCRARE
(
id_lucrare int,
nume varchar (50),
cod_locatie int not null,
cod_sef int not null,
constraint locatie_fk_lucrare foreign key (cod_locatie) references LOCATIE(id_locatie),
constraint sef_fk_lucrare foreign key (cod_sef) references SEF(id_sef)
);

alter table LUCRARE
add constraint id_lucrare_pk primary key(id_lucrare);

create table FACTURA
(
id_factura int,
cod_oferta_acceptata int,
cod_client int,
cod_lucrare int not null,
data_facturare date not null,
constraint factura_fk_oferta foreign key (cod_oferta_acceptata) references OFERTA(id_oferta),
constraint factura_fk_lucrarea foreign key (cod_lucrare) references LUCRARE(id_lucrare),
constraint factura_fk_client foreign key (cod_client) references CLIENT(id_client)
);

alter table FACTURA
add constraint id_factura_pk primary key(id_factura);

create table RECENZIE
(
id_recenzie int,
cod_client int not null,
cod_lucrare int not null,
parere varchar(200),
constraint recenzie_fk_lucrare foreign key (cod_lucrare) references LUCRARE(id_lucrare),
constraint recenzie_fk_client foreign key (cod_client) references CLIENT(id_client)
);

alter table RECENZIE
add constraint id_recenzie_pk primary key(id_recenzie);

create table ECHIPA
(
id_echipa int,
cod_sef int not null,
constraint echipa_fk_sef foreign key (cod_sef) references SEF(id_sef)
);
alter table ECHIPA
add constraint id_echipa_pk primary key(id_echipa);

create table ANGAJAT
(
id_angajat int,
nume varchar(100) not null, 
prenume varchar(100) not null,
numar_telefon varchar(15) not null, 
varsta  int constraint varsta_validul check(varsta>=18),
specializare varchar(30),
experienta int constraint experienta_validul check(experienta>=0), 
salariu int constraint salariu_validul check(salariu>0),
cod_echipa int default NULL,
constraint angajat_fk_echipa foreign key (cod_echipa) references ECHIPA(id_echipa)
);

alter table ANGAJAT
add constraint id_angajat_pk primary key(id_angajat);

create table PROGRAMARE_UTILAJE
(
cod_utilaj int not null,
cod_locatie int not null,
data_inceput date not null,
data_terminare date not null,
constraint programare_pk_utilaje primary key (cod_utilaj,cod_locatie),
constraint programare_fk_utilaj foreign key (cod_utilaj) references UTILAJ(id_utilaj),
constraint programare_fk_locatie foreign key (cod_locatie) references LOCATIE(id_locatie)
);

create table SELECTIE_COLABORATORI
(
id_selectie int not null,
cod_colaborator int not null,
cod_oferta int not null,
constraint selectie_pk_colaboratori primary key (id_selectie),
constraint selectie_fk_colaboratori foreign key (cod_colaborator) references COLABORATORI(id_colaborator),
constraint selectie_fk_colaboratori2 foreign key (cod_oferta) references OFERTA(id_oferta)
);

SELECT *
FROM ADMIN_APLICATIE.LOCATIE
WHERE localitate = 'Bucuresti';

ADD CONSTRAINT PK_LOCATIE_BUCURESTI
PRIMARY KEY (id_locatie);

FROM LOCATIE_BUCURESTI;

--Fragmente orizontale derivate
--lucrare
CREATE TABLE LUCRARE_BUCURESTI AS
SELECT l.*
FROM ADMIN_APLICATIE.LUCRARE l
JOIN LOCATIE_BUCURESTI loc
  ON l.cod_locatie = loc.id_locatie;

ALTER TABLE LUCRARE_BUCURESTI
ADD CONSTRAINT PK_LUCRARE_BUCURESTI
PRIMARY KEY (id_lucrare);

--factura
CREATE TABLE FACTURA_BUCURESTI AS
SELECT f.*
FROM ADMIN_APLICATIE.FACTURA f
JOIN LUCRARE_BUCURESTI l
  ON f.cod_lucrare = l.id_lucrare;

ALTER TABLE FACTURA_BUCURESTI
ADD CONSTRAINT PK_FACTURA_BUCURESTI
PRIMARY KEY (id_factura);

--recenzie
CREATE TABLE RECENZIE_BUCURESTI AS
SELECT r.*
FROM ADMIN_APLICATIE.RECENZIE r
JOIN LUCRARE_BUCURESTI l
  ON r.cod_lucrare = l.id_lucrare;

ALTER TABLE RECENZIE_BUCURESTI
ADD CONSTRAINT PK_RECENZIE_BUCURESTI
PRIMARY KEY (id_recenzie);

--programare_utilaje

CREATE TABLE PROGRAMARE_UTILAJE_BUCURESTI AS
SELECT p.*
FROM ADMIN_APLICATIE.PROGRAMARE_UTILAJE p
JOIN LOCATIE_BUCURESTI loc
  ON p.cod_locatie = loc.id_locatie;
  
ALTER TABLE PROGRAMARE_UTILAJE_BUCURESTI
ADD CONSTRAINT PK_PROG_UTIL_BUC
PRIMARY KEY (cod_utilaj, cod_locatie);

--5. Asigurarea sincronizarii datelor pentru relatiile replicate
--Trigger replicare CLIENT
CREATE OR REPLACE TRIGGER TRG_REPL_CLIENT_BUC
AFTER INSERT OR UPDATE OR DELETE ON CLIENT_BUCURESTI
FOR EACH ROW
BEGIN
  IF INSERTING THEN
    INSERT INTO CLIENT_NON_BUCURESTI@bd_non_bucuresti
    (id_client, nume, prenume, numar_telefon, nr_lucrari_anterioare)
    VALUES
    (:NEW.id_client, :NEW.nume, :NEW.prenume,
     :NEW.numar_telefon, :NEW.nr_lucrari_anterioare);

  ELSIF UPDATING THEN
    UPDATE CLIENT_NON_BUCURESTI@bd_non_bucuresti
    SET nume = :NEW.nume,
        prenume = :NEW.prenume,
        numar_telefon = :NEW.numar_telefon,
        nr_lucrari_anterioare = :NEW.nr_lucrari_anterioare
    WHERE id_client = :OLD.id_client;

  ELSIF DELETING THEN
    DELETE FROM CLIENT_NON_BUCURESTI@bd_non_bucuresti
    WHERE id_client = :OLD.id_client;
  END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_CHECK_ANG_SENS
BEFORE INSERT OR UPDATE ON ANGAJAT_DATE_SENSIBILE
FOR EACH ROW
DECLARE
  v_count NUMBER;
BEGIN
  SELECT COUNT(*)
  INTO v_count
  FROM ANGAJAT_DATE_PROFESIONALE@bd_bucuresti
  WHERE id_angajat = :NEW.id_angajat;

  IF v_count = 0 THEN
    RAISE_APPLICATION_ERROR(
      -20001,
      'Nu exista angajatul in fragmentul profesional.'
    );
  END IF;
END;
